-- ===================================================== -- QC HISTORY TABLE WITH AUTO-ROTATION PARTITIONING -- Untuk Supabase (PostgreSQL) -- ===================================================== -- 1. Drop tabel jika sudah ada (untuk fresh install) DROP TABLE IF EXISTS qc_history CASCADE; DROP FUNCTION IF EXISTS public.auto_create_qc_partition() CASCADE; -- 2. Buat tabel utama dengan partitioning berdasarkan ID CREATE TABLE IF NOT EXISTS qc_history ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(), username VARCHAR(100) NOT NULL, location VARCHAR(255) NOT NULL, status VARCHAR(50) NOT NULL CHECK (status IN ('OKE', 'REVISI', 'REJECT')), jo_number VARCHAR(100) NOT NULL, product_code VARCHAR(100) NOT NULL, product_name VARCHAR(255) NOT NULL, serial_number TEXT, reason VARCHAR(255) DEFAULT '-', total_qty INTEGER NOT NULL CHECK (total_qty > 0), checked_qty INTEGER NOT NULL CHECK (checked_qty >= 0), remarks TEXT, timestamp VARCHAR(100) ) PARTITION BY RANGE (id); -- 3. Buat partition pertama (ID 1 - 1.000.000) -- Setiap partition menampung 1 juta record CREATE TABLE IF NOT EXISTS qc_history_1 PARTITION OF qc_history FOR VALUES FROM (1) TO (1000001); -- 4. Fungsi untuk otomatis membuat partition baru CREATE OR REPLACE FUNCTION public.auto_create_qc_partition() RETURNS TRIGGER AS $$ DECLARE v_partition_num INTEGER; v_partition_name TEXT; v_start_id BIGINT; v_end_id BIGINT; BEGIN -- Hitung nomor partition berdasarkan ID -- Contoh: ID 1.000.001 akan masuk ke partition 2 v_partition_num := (NEW.id - 1) / 1000000 + 1; v_partition_name := 'qc_history_' || v_partition_num; v_start_id := (v_partition_num - 1) * 1000000 + 1; v_end_id := v_partition_num * 1000000 + 1; -- Cek apakah partition sudah ada IF NOT EXISTS (SELECT 1 FROM pg_class WHERE relname = v_partition_name AND relkind = 'r') THEN -- Buat partition baru EXECUTE format( 'CREATE TABLE IF NOT EXISTS %I PARTITION OF qc_history FOR VALUES FROM (%L) TO (%L)', v_partition_name, v_start_id, v_end_id ); -- Log untuk monitoring (opsional) RAISE NOTICE 'Created new partition: % for IDs % to %', v_partition_name, v_start_id, v_end_id - 1; END IF; RETURN NEW; END; $$ LANGUAGE plpgsql SECURITY DEFINER; -- 5. Pasang trigger untuk auto-creation partition DROP TRIGGER IF EXISTS trg_auto_create_qc_partition ON qc_history; CREATE TRIGGER trg_auto_create_qc_partition BEFORE INSERT ON qc_history FOR EACH ROW EXECUTE FUNCTION public.auto_create_qc_partition(); -- 6. Index untuk performa query (dibuat otomatis di setiap partition) CREATE INDEX IF NOT EXISTS idx_qc_history_jo_number ON qc_history(jo_number); CREATE INDEX IF NOT EXISTS idx_qc_history_product_code ON qc_history(product_code); CREATE INDEX IF NOT EXISTS idx_qc_history_location ON qc_history(location); CREATE INDEX IF NOT EXISTS idx_qc_history_created_at ON qc_history(created_at DESC); CREATE INDEX IF NOT EXISTS idx_qc_history_status ON qc_history(status); -- ===================================================== -- PANDUAN PENGGUNAAN -- ===================================================== -- -- 1. Setiap partition menampung 1.000.000 record -- 2. Partition dibuat otomatis ketika data baru di-insert -- 3. Tabel lama (partition 1, 2, dst) tetap bisa diakses -- 4. Data baru otomatis masuk ke partition yang sesuai berdasarkan ID -- 5. Untuk mengubah kapasitas per partition, ubah angka 1000000 di fungsi -- -- Contoh query untuk monitoring: -- SELECT schemaname, tablename, pg_size_pretty(pg_total_relation_size(schemaname||'.'||tablename)) AS size -- FROM pg_tables WHERE tablename LIKE 'qc_history_%' ORDER BY pg_total_relation_size(schemaname||'.'||tablename) DESC; -- -- Contoh query untuk cek jumlah record per partition: -- SELECT -- 'qc_history_' || ((id - 1) / 1000000 + 1) as partition_name, -- COUNT(*) as record_count -- FROM qc_history -- GROUP BY partition_name -- ORDER BY partition_name; -- =====================================================
